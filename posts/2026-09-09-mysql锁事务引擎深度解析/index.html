MySQL 锁、事务、引擎深度解析

核心认知：MySQL 的「锁」不是单一概念，而是 全局锁 → 表锁 → 行锁 → 意向锁 → 间隙锁 → Next-Key Lock 的层级防御体系；MVCC 不是魔法，是 Undo Log + ReadView + 隐式主键 的精密协作；死锁不是bug，是高并发下的必然概率事件，关键是「如何快速发现、如何优雅恢复」。

二、行锁实现原理：从索引到 Lock Struct

2.1 核心数据结构

// InnoDB 锁结构 (简化自 lock0lock.h)
struct lock_t {
    trx_t*      trx;              // 所属事务
    dict_index_t* index;          // 所属索引
    lock_mode_t mode;             // LOCK_X | LOCK_S | LOCK_GAP | LOCK_ORDINARY | LOCK_INSERT_INTENTION
    lock_type_t type;             // LOCK_REC | LOCK_TABLE
    ulint       hash;             // Hash 值 (space_id + page_no + n_bits)
    bitmap_t    bitmap;           // 位图：每 bit 对应一条记录 (heap_no)
    // 链表指针
    lock_t*     trx_locks_next;   // 事务锁链表
    lock_t*     trx_locks_prev;
};

// 记录锁位图映射：heap_no -> bit
// heap_no: 页内记录偏移 (从 0 开始，0=infimum, 1=supremum, 2+=用户记录)

2.2 行锁加锁流程源码逻辑

// row0sel.cc: row_search_for_mysql() -> 行锁加锁入口
// 核心函数：lock_rec_lock()

bool lock_rec_lock(
    lock_mode_t mode,        // 锁模式
    dict_index_t* index,     // 索引
    const ulint* rec,        // 记录指针
    lock_type_t type,        // LOCK_REC
    trx_t* trx,              // 事务
    ulint n_bits             // 位图长度
) {
    // 1. 计算 hash，定位 Lock Sys Hash 表桶
    ulint hash = lock_rec_hash(space_id, page_no, n_bits);
    
    // 2. 遍历桶内链表，查找是否已有同页锁对象
    for (lock_t* lock = HASH_GET(list, hash); lock; lock = lock->hash_next) {
        if (lock->index == index && lock->space == space_id && lock->page_no == page_no) {
            // 3. 找到现有锁对象，设置对应 heap_no 的 bit
            lock_rec_add_to_queue(lock, mode, heap_no);
            return true;  // 成功
        }
    }
    
    // 4. 未找到，创建新 lock_t，分配 bitmap，设置 bit
    lock_t* lock = lock_create(...);
    lock_rec_add_to_queue(lock, mode, heap_no);
    
    // 5. 插入 Hash 表 & 事务锁链表
    HASH_INSERT(list, hash, lock);
    trx_lock_add(lock, trx);
    
    return true;
}

2.3 关键点：heap_no 与位图

数据页结构 (16KB)：
┌─────────────────────────────────────┐
│ Page Header (56 bytes)              │
├─────────────────────────────────────┤
│ Infimum Record (heap_no=0)          │  系统最小记录
├─────────────────────────────────────┤
│ User Record 1 (heap_no=2)           │  实际数据
├─────────────────────────────────────┤
│ User Record 2 (heap_no=3)           │
├─────────────────────────────────────┤
│ ...                                 │
├─────────────────────────────────────┤
│ Supremum Record (heap_no=1)         │  系统最大记录
├─────────────────────────────────────┤
│ Page Directory (稀疏索引)           │
├─────────────────────────────────────┤
│ Free Space                          │
└─────────────────────────────────────┘

Lock Bitmap (每 bit 对应一个 heap_no)：
bit 0: Infimum  (间隙锁左边界)
bit 1: Supremum (间隙锁右边界)
bit 2: Record 1
bit 3: Record 2
...

Next-Key Lock 本质：锁住 记录本身 + 记录前的间隙

  • 对应 bitmap：bit[heap_no] (记录) + bit[heap_no-1] (前间隙)

三、MVCC 多版本并发控制：读不加锁的秘密

3.1 核心组件

MVCC 核心三件套：

│  Undo Log       │  │  ReadView       │  │  Hidden Columns │

3.2 隐式列与版本链

-- 每行记录隐藏列 (不显示在 SHOW CREATE TABLE)
-- 可通过 SELECT * FROM information_schema.innodb_sys_tables 间接验证

版本链演示：

-- 更新 2

  DB_TRX_ID=300, DB_ROLL_PTR=0x7f...  ───►

*/

3.3 ReadView 可见性判断算法

// 源码: read0read.cc: read_view_sees_trx_id()
// ReadView 结构
struct ReadView {
    trx_id_t   low_limit_id;    // 活跃事务最小 ID (未提交最小)
    trx_id_t   up_limit_id;     // 已提交最大 ID + 1 (高水位)
    trx_id_t*  trx_ids;         // 活跃事务 ID 列表 (排序)
    ulint      n_trx_ids;
    trx_id_t   creator_trx_id;  // 创建者事务 ID
};

// 可见性判断核心逻辑
bool trx_id_is_visible(trx_id_t trx_id, ReadView* view) {
    // 1. 自己修改的，可见
    if (trx_id == view->creator_trx_id) return true;
    
    // 2. 比已提交最大还小，可见
    if (trx_id < view->up_limit_id) return true;
    
    // 3. 比活跃最小还大，不可见 (未提交)
    if (trx_id >= view->low_limit_id) return false;
    
    // 4. 在活跃列表中，不可见 (未提交)
    // 二分查找 trx_ids 数组
    if (binary_search(view->trx_ids, view->n_trx_ids, trx_id)) {
        return false;
    }
    
    // 5. 不在活跃列表，且 < low_limit_id，可见 (已提交)
    return true;
}

3.4 RC vs RR 隔离级别的 ReadView 生命周期

隔离级别ReadView 创建时机可见性差异
READ COMMITTED (RC)每个 SELECT 语句开始每次查询看到最新已提交数据，可能不可重复读
REPEATABLE READ (RR)事务第一个 SELECT事务内所有 SELECT 共享同一 ReadView，可重复读
-- RC 场景：不可重复读
-- 事务 A                    -- 事务 B
START TRANSACTION;           START TRANSACTION;
SELECT * FROM t WHERE id=1;  -- 'Alice'
                             UPDATE t SET name='Bob' WHERE id=1;
                             COMMIT;
SELECT * FROM t WHERE id=1;  -- 'Bob' (RC: 不同！RR: 仍是 'Alice')
COMMIT;

3.5 MVCC 与锁的配合：一致性非锁定读 vs 锁定读

-- 一致性非锁定读 (普通 SELECT) - MVCC，不加锁
SELECT * FROM t WHERE id = 1;

-- 锁定读 (显式加锁)
SELECT * FROM t WHERE id = 1 LOCK IN SHARE MODE;      -- S 锁 (RC/RR: 记录锁)
SELECT * FROM t WHERE id = 1 FOR UPDATE;              -- X 锁 (RC/RR: 记录锁)
SELECT * FROM t WHERE id = 1 FOR UPDATE SKIP LOCKED;  -- 跳过已锁行 (JDK 8+)
SELECT * FROM t WHERE id = 1 FOR UPDATE NOWAIT;       -- 立即失败不等待 (JDK 8+)
SELECT * FROM t WHERE id = 1 FOR SHARE SKIP LOCKED;   -- S 锁跳过

四、事务隔离级别与实现差异

4.1 四大隔离级别对比

隔离级别脏读不可重复读幻读InnoDB 实现
READ UNCOMMITTED✓✓✓直接读最新版本，无 ReadView
READ COMMITTED✗✓✓每次 SELECT 生成 ReadView
REPEATABLE READ✗✗✗*事务首次 SELECT 生成 ReadView + Next-Key Lock
SERIALIZABLE✗✗✗读加 S 锁，写加 X 锁，串行化

*InnoDB RR 通过 Next-Key Lock 解决幻读，但不完全等同于串行化（如范围查询+插入仍可能并发）。

4.2 幻读在 RR 下的解决：Next-Key Lock

-- 事务 A (RR)
START TRANSACTION;
SELECT * FROM t WHERE id > 10 FOR UPDATE;  -- 加 Next-Key Lock
-- 锁住：(10, +∞) 间隙 + 所有 id>10 的记录

-- 事务 B
INSERT INTO t VALUES (15, 'x');  -- 阻塞！等待间隙锁释放

Next-Key Lock 加锁规则（RR 默认）：

  1. 唯一索引等值查询 → Record Lock (退化)
  2. 唯一索引范围查询 → Next-Key Lock
  3. 非唯一索引等值查询 → Next-Key Lock
  4. 非唯一索引范围查询 → Next-Key Lock
  5. 无索引全表扫描 → 表锁 (退化，性能灾难)

五、死锁：原理、复现、排查、预防

5.1 死锁必要条件（Coffman 条件）

  1. 互斥：资源独占
  2. 持有并等待：持有锁等待新锁
  3. 不可抢占：锁只能主动释放
  4. 循环等待：环形等待链

5.2 经典死锁复现：两事务反向更新

-- 表结构
CREATE TABLE account (
    id INT PRIMARY KEY,
    balance DECIMAL(10,2)
);
INSERT INTO account VALUES (1, 100), (2, 100);

-- 事务 A                        -- 事务 B
START TRANSACTION;               START TRANSACTION;
UPDATE account SET balance=balance-10 WHERE id=1;  -- 锁 id=1
                                    UPDATE account SET balance=balance-10 WHERE id=2;  -- 锁 id=2
UPDATE account SET balance=balance+10 WHERE id=2;  -- 等待 id=2 (B持有)
                                    UPDATE account SET balance=balance+10 WHERE id=1;  -- 等待 id=1 (A持有)
-- DEADLOCK!                      -- DEADLOCK!

八、线上典型事故案例复盘

案例 1：无索引 UPDATE 导致全表锁

-- 事故 SQL
UPDATE user SET status=1 WHERE phone='13800000000';  -- phone 无索引

-- 现象：全库阻塞，所有写入等待
-- 原因：无索引 → 全表扫描 → 每行加 Next-Key Lock → 退化表锁
-- 解决：加索引 `idx_phone(phone)`

案例 2：大事务导致 Undo Log 爆盘

-- 事故：一次性删除 500 万行
DELETE FROM logs WHERE create_time < '2025-01-01';

-- 后果：
-- 1. Undo Log 暴涨 50GB+
-- 2. Purge 线程跟不上，History List Length 飙升
-- 3. 后续 SELECT 因遍历版本链极慢
-- 4. 主从延迟严重

-- 解决：分批删除 (每次 5000 行 + 间隔 sleep)

案例 3：隐式转换导致索引失效

-- 表：id VARCHAR(20) PRIMARY KEY
-- 事故：
SELECT * FROM t WHERE id = 12345;  -- 字符串列比数字，隐式转换 CAST(id AS UNSIGNED)

-- 现象：全表扫描，锁全表
-- 解决：WHERE id = '12345' 或 WHERE CAST(id AS CHAR) = '12345'

案例 4：Gap Lock 导致插入死锁

-- 事务 A
SELECT * FROM t WHERE id > 100 FOR UPDATE;  -- 锁 (100, +∞)

-- 事务 B
INSERT INTO t VALUES (200, 'x');  -- 等待 Gap Lock

-- 事务 A 提交后，B 成功
-- 但高并发下大量插入排队，RT 飙升

-- 优化：改用 RC 隔离级别 (无 Gap Lock) 或精确索引定位

十、12 年经验总结：MySQL 锁事务避坑清单

□ 所有表必须有显式主键 (推荐 BIGINT AUTO_INCREMENT)
□ 所有 WHERE/JOIN/ORDER BY 列建立索引 (EXPLAIN 验证)
□ 禁止 SELECT *，显式列出列名
□ 事务尽可能短，不在事务中调用 RPC/HTTP
□ 批量操作分批 (500-1000 行/批)
□ 统一多表更新顺序 (按表名/主键排序)
□ 显式指定隔离级别 (RC + ROW binlog 通用最优)
□ 监控：Innodb_row_lock_waits, Innodb_deadlocks, History_list_length
□ 开启 innodb_print_all_deadlocks = ON
□ 定期 PURGE 大事务表 (pt-archiver / 分区表)
□ 备库同步延迟监控 (Seconds_Behind_Master / GTID)
□ 备份验证：定期恢复演练

十一、核心源码阅读路线图

1. 事务系统
   trx0trx.cc: trx_t 结构、事务状态机、提交/回滚

2. 锁系统
   lock0lock.cc: lock_t 结构、加锁/释放、死锁检测
   lock0types.h: 锁模式位掩码定义

3. MVCC
   read0read.cc: ReadView 创建、可见性判断、purge
   undo0undo.cc: Undo Log 写入、版本链维护

4. 索引与行锁
   row0sel.cc: 行搜索、加锁入口
   btr0cur.cc: B+Tree 游标操作

5. Buffer Pool
   buf0buf.cc: 页管理、LRU、Flush
   buf0flu.cc: 刷盘策略

6. Redo Log
   log0log.cc: Redo 写入、Checkpoint、恢复
